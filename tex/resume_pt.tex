% Render a comma-separated list of tags (optionally "text/true") as \cvtag pills
\newcommand{\cvtags}[1]{%
  \foreach \tagitem in {#1} {%
    \IfSubStr{\tagitem}{/}%
      {\StrBefore{\tagitem}{/}[\tagname]}%
      {\def\tagname{\tagitem}}%
    \cvtag{\tagname}\ %
  }%
}
